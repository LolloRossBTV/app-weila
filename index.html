import streamlit as st import datetime import pandas as pd # 1. Configurazione e Ottimizzazione Layout (Smartphone-friendly per clienti e staff) st.set_page_config(page_title="Gestione Weilà & Kai", layout="centered") st.markdown(""" """, unsafe_allow_html=True) # 2. Simulazione Database in memoria (Utenti, Menu, Comande) if 'utenti' not in st.session_state: st.session_state.utenti = { "admin": {"pwd": "123", "ruolo": "Admin"}, "mario": {"pwd": "123", "ruolo": "Senior"}, "luigi": {"pwd": "123", "ruolo": "Junior"} } if 'menu' not in st.session_state: st.session_state.menu = [ {"id": 1, "nome": "Spaghetti all'Assassina", "prezzo": 12.00, "categoria": "Primi"}, {"id": 2, "nome": "Pizza Weilà", "prezzo": 13.00, "categoria": "Pizze"}, {"id": 3, "nome": "Acqua Naturale", "prezzo": 2.00, "categoria": "Bevande"} ] if 'comande' not in st.session_state: st.session_state.comande = [] # Memorizza {id_ordine, tavolo, totale, stato, piatti} if 'carrello' not in st.session_state: st.session_state.carrello = [] if 'utente_loggato' not in st.session_state: st.session_state.utente_loggato = None # 3. Instradamento tramite URL tavolo = st.query_params.get("tavolo", None) # ========================================== # LATO CLIENTE (Se c'è ?tavolo nell'URL) # ========================================== if tavolo: st.title(f"📍 Tavolo {tavolo}") st.caption("Benvenuti da Weilà & Kai. Componi il tuo ordine.") # Visualizzazione Menu categorie = sorted(list(set([p["categoria"] for p in st.session_state.menu]))) for cat in categorie: st.subheader(cat) for piatto in [p for p in st.session_state.menu if p["categoria"] == cat]: col1, col2 = st.columns([3, 1]) with col1: st.write(f"**{piatto['nome']}** - €{piatto['prezzo']:.2f}") with col2: if st.button("➕", key=f"add_{piatto['id']}"): st.session_state.carrello.append(piatto) st.toast(f"{piatto['nome']} aggiunto!") st.divider() # Carrello e Invio Ordine st.subheader("🛒 Il tuo Ordine") if not st.session_state.carrello: st.info("Il carrello è vuoto.") else: for i, item in enumerate(st.session_state.carrello): st.write(f"- {item['nome']}") totale = sum(item['prezzo'] for item in st.session_state.carrello) st.write(f"**Totale: €{totale:.2f}**") if st.button("🚀 INVIA ORDINE IN CUCINA", type="primary"): nuova_comanda = { "id_ordine": len(st.session_state.comande) + 1, "tavolo": tavolo, "piatti": [p["nome"] for p in st.session_state.carrello], "totale": totale, "stato": "In Attesa", "ora": datetime.datetime.now().strftime("%H:%M") } st.session_state.comande.append(nuova_comanda) st.session_state.carrello = [] # Svuota carrello st.success("Ordine inviato con successo!") st.rerun() # ========================================== # LATO STAFF (Se il link è pulito) # ========================================== else: # 4. Sistema di Login if not st.session_state.utente_loggato: st.title("🔐 Accesso Staff") with st.form("login_form"): username = st.text_input("Username") password = st.text_input("Password", type="password") submit = st.form_submit_button("Accedi") if submit: if username in st.session_state.utenti and st.session_state.utenti[username]["pwd"] == password: st.session_state.utente_loggato = st.session_state.utenti[username] st.session_state.utente_loggato["username"] = username st.rerun() else: st.error("Credenziali non valide.") # 5. Dashboard Protetta (Gestione basata sui Ruoli) else: user = st.session_state.utente_loggato st.sidebar.title(f"👤 {user['username'].capitalize()}") st.sidebar.caption(f"Ruolo: {user['ruolo']}") if st.sidebar.button("🚪 Esci"): st.session_state.utente_loggato = None st.rerun() st.title("Pannello di Controllo") # PANNELLO COMUNE: Gestione Comande (Visibile a Junior, Senior, Admin) st.header("📋 Comande in Arrivo") if not st.session_state.comande: st.info("Nessuna comanda attiva.") else: for comanda in reversed(st.session_state.comande): with st.expander(f"Tavolo {comanda['tavolo']} - {comanda['ora']} ({comanda['stato']})"): st.write(f"**Piatti:** {', '.join(comanda['piatti'])}") st.write(f"**Totale:** €{comanda['totale']:.2f}") # I Junior possono solo gestire lo stato, Senior/Admin anche annullare col1, col2 = st.columns(2) if comanda['stato'] == "In Attesa": if col1.button("✅ Metti in lavorazione", key=f"lav_{comanda['id_ordine']}"): comanda['stato'] = "In Lavorazione" st.rerun() elif comanda['stato'] == "In Lavorazione": if col1.button("🏁 Segna Evato", key=f"eva_{comanda['id_ordine']}"): comanda['stato'] = "Evato" st.rerun() if user['ruolo'] in ["Senior", "Admin"]: if col2.button("❌ Annulla Comanda", key=f"ann_{comanda['id_ordine']}"): st.session_state.comande.remove(comanda) st.rerun() st.divider() # PANNELLO SENIOR E ADMIN: Modifica Menu if user['ruolo'] in ["Senior", "Admin"]: st.header("🍕 Gestione Menu") with st.form("aggiungi_piatto"): col1, col2, col3 = st.columns(3) nome = col1.text_input("Nome Piatto") prezzo = col2.number_input("Prezzo €", min_value=0.0, step=0.5) categoria = col3.text_input("Categoria") if st.form_submit_button("Aggiungi Piatto"): nuovo_id = max([p['id'] for p in st.session_state.menu]) + 1 if st.session_state.menu else 1 st.session_state.menu.append({"id": nuovo_id, "nome": nome, "prezzo": prezzo, "categoria": categoria}) st.success("Piatto aggiunto! I clienti lo vedranno immediatamente.") st.rerun() st.dataframe(pd.DataFrame(st.session_state.menu)) # PANNELLO ADMIN: Gestione Utenti e Statistiche if user['ruolo'] == "Admin": st.divider() st.header("⚙️ Amministrazione") st.write("Utenti registrati nel sistema:") st.json(st.session_state.utenti)
